export class PayrollEngine {
  public async calculateMonthlySalary(employeeId: string) {
    const employee = await db.employees.findById(employeeId);
    
    // Hitung pemotongan pajak PPh21 dan insentif
    const taxDeduction = calculateTax(employee.baseSalary);
    const netPay = employee.baseSalary - taxDeduction + employee.bonus;

    return await this.generateEncryptedPayslip({
      employeeId,
      netPay,
      issuedAt: Date.now()
    });
  }
}

Ekosistem penggajian terpusat yang memangkas waktu kalkulasi bulanan hingga 90% dengan keandalan tinggi.
